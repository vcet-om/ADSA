Q1 AVL Tree Construction

class Node {
    int key, height;
    Node left, right;
    Node(int d) { key = d; height = 1; }
}
public class AVLTree {
    Node root;
    int height(Node N) { return (N == null) ? 0 : N.height; }
    int max(int a, int b) { return (a > b) ? a : b; }
    Node rightRotate(Node y) {
        Node x = y.left;
        Node T2 = x.right;
        x.right = y;
        y.left = T2;
        y.height = max(height(y.left), height(y.right)) + 1;
        x.height = max(height(x.left), height(x.right)) + 1;
        return x;
    }
    Node leftRotate(Node x) {
        Node y = x.right;
        Node T2 = y.left;
        y.left = x;
        x.right = T2;
        x.height = max(height(x.left), height(x.right)) + 1;
        y.height = max(height(y.left), height(y.right)) + 1;
        return y;
    }
    int getBalance(Node N) { return (N == null) ? 0 : height(N.left) - height(N.right); }
    Node insert(Node node, int key) {
        if (node == null) return (new Node(key));
        if (key < node.key) node.left = insert(node.left, key);
        else if (key > node.key) node.right = insert(node.right, key);
        else return node;
        node.height = 1 + max(height(node.left), height(node.right));
        int balance = getBalance(node);
        if (balance > 1 && key < node.left.key) return rightRotate(node);
        if (balance < -1 && key > node.right.key) return leftRotate(node);
        if (balance > 1 && key > node.left.key) {
            node.left = leftRotate(node.left);
            return rightRotate(node);
        }
        if (balance < -1 && key < node.right.key) {
            node.right = rightRotate(node.right);
            return leftRotate(node);
        }
        return node;
    }
    void preOrder(Node node) {
        if (node != null) {
            System.out.print(node.key + " ");
            preOrder(node.left);
            preOrder(node.right);
        }
    }
    public static void main(String[] args) {
        AVLTree tree = new AVLTree();
        int[] keys = {50, 20, 70, 10, 30, 60, 80, 25};
        for (int key : keys) {
            tree.root = tree.insert(tree.root, key);
        }
        System.out.println("Preorder traversal of constructed AVL tree:");
        tree.preOrder(tree.root);
        System.out.println();
    }
}

📋 Program Output

Preorder traversal of constructed AVL tree: 50 20 10 30 25 70 60 80

How It Works

This program constructs an AVL Tree by inserting keys: 50, 20, 70, 10, 30, 60, 80, 25. AVL trees are self-balancing BSTs where the difference between heights of left and right subtrees (balance factor) is at most 1.

  • rightRotate() and leftRotate() rebalance the tree when insertions cause imbalance
  • getBalance() computes the balance factor of any node
  • Four rotation cases are handled: LL, RR, LR, and RL
Time Complexity: O(log n) per insertion

import java.util.*;

class BTree {
    private int order;
    private Node root;
    public BTree(int order) { this.order = order; this.root = new Node(true); }
    class Node {
        List<Integer> keys;
        List<Node> children;
        boolean leaf;
        Node(boolean leaf) { this.leaf = leaf; keys = new ArrayList<>(); children = new ArrayList<>(); }
    }
    public void insert(int key) {
        Node r = root;
        if (r.keys.size() == order - 1) {
            Node s = new Node(false);
            root = s;
            s.children.add(r);
            splitChild(s, 0, r);
            insertNonFull(s, key);
        } else { insertNonFull(r, key); }
    }
    private void splitChild(Node x, int i, Node y) {
        Node z = new Node(y.leaf);
        int mid = (order - 1) / 2;
        int keyUp = y.keys.get(mid);
        for (int j = mid + 1; j < y.keys.size(); j++) z.keys.add(y.keys.get(j));
        if (!y.leaf) for (int j = mid + 1; j < y.children.size(); j++) z.children.add(y.children.get(j));
        y.keys.subList(mid, y.keys.size()).clear();
        if (!y.leaf) y.children.subList(mid + 1, y.children.size()).clear();
        x.children.add(i + 1, z);
        x.keys.add(i, keyUp);
    }
    private void insertNonFull(Node x, int key) {
        int i = x.keys.size() - 1;
        if (x.leaf) {
            x.keys.add(0);
            while (i >= 0 && x.keys.get(i) > key) { x.keys.set(i + 1, x.keys.get(i)); i--; }
            x.keys.set(i + 1, key);
        } else {
            while (i >= 0 && x.keys.get(i) > key) i--;
            i++;
            if (x.children.get(i).keys.size() == order - 1) {
                splitChild(x, i, x.children.get(i));
                if (key > x.keys.get(i)) i++;
            }
            insertNonFull(x.children.get(i), key);
        }
    }
    public void printTree(Node x, int level) {
        System.out.print("Level " + level + ": ");
        for (int k : x.keys) System.out.print(k + " ");
        System.out.println();
        if (!x.leaf) for (Node c : x.children) printTree(c, level + 1);
    }
    public void print() { printTree(root, 0); }
}
public class BTreeOrder3 {
    public static void main(String[] args) {
        BTree tree = new BTree(3);
        int[] keys = {10, 20, 5, 6, 12, 30, 7, 17};
        for(int k : keys) tree.insert(k);
        System.out.println("B-Tree of order 3 after inserting elements:");
        tree.print();
    }
}

How It Works

A B-Tree of order 3 (2-3 tree) allows at most 2 keys and 3 children per node. Keys inserted: 10, 20, 5, 6, 12, 30, 7, 17.

  • When a node overflows (reaches 2 keys), it splits and pushes the middle key up
  • splitChild() handles the splitting logic
  • insertNonFull() finds the correct leaf and inserts
Time Complexity: O(log n) per insertion

import java.util.*;

class BPlusTree {
    private int order;
    private Node root;
    public BPlusTree(int order) { this.order = order; this.root = new Node(true); }
    class Node {
        List<Integer> keys; List<Node> children; boolean leaf; Node next;
        Node(boolean leaf) { this.leaf = leaf; keys = new ArrayList<>(); children = new ArrayList<>(); }
    }
    public void insert(int key) {
        Node r = root;
        if (r.keys.size() == order - 1) {
            Node s = new Node(false); root = s; s.children.add(r);
            splitChild(s, 0, r); insertNonFull(s, key);
        } else { insertNonFull(r, key); }
    }
    private void splitChild(Node x, int i, Node y) {
        Node z = new Node(y.leaf); int mid = (order - 1) / 2;
        if (y.leaf) {
            for (int j = mid; j < y.keys.size(); j++) z.keys.add(y.keys.get(j));
            y.keys.subList(mid, y.keys.size()).clear();
            z.next = y.next; y.next = z;
            x.children.add(i + 1, z); x.keys.add(i, z.keys.get(0));
        } else {
            int keyUp = y.keys.get(mid);
            for (int j = mid + 1; j < y.keys.size(); j++) z.keys.add(y.keys.get(j));
            for (int j = mid + 1; j < y.children.size(); j++) z.children.add(y.children.get(j));
            y.keys.subList(mid, y.keys.size()).clear();
            y.children.subList(mid + 1, y.children.size()).clear();
            x.children.add(i + 1, z); x.keys.add(i, keyUp);
        }
    }
    private void insertNonFull(Node x, int key) {
        int i = x.keys.size() - 1;
        if (x.leaf) {
            x.keys.add(0); while (i >= 0 && x.keys.get(i) > key) { x.keys.set(i + 1, x.keys.get(i)); i--; }
            x.keys.set(i + 1, key);
        } else {
            while (i >= 0 && x.keys.get(i) > key) i--; i++;
            if (x.children.get(i).keys.size() == order - 1) {
                splitChild(x, i, x.children.get(i)); if (key > x.keys.get(i)) i++;
            }
            insertNonFull(x.children.get(i), key);
        }
    }
    public void printTree(Node x, int level) {
        System.out.println("Level " + level + ": " + x.keys);
        if (!x.leaf) for (Node c : x.children) printTree(c, level + 1);
    }
    public void print() { printTree(root, 0); }
    public void printLeaves() {
        Node curr = root; while (!curr.leaf) curr = curr.children.get(0);
        System.out.print("Leaf nodes (linked list): ");
        while (curr != null) { System.out.print(curr.keys + " -> "); curr = curr.next; }
        System.out.println("null");
    }
}

public class BFSGraph {
    public static void main(String[] args) {
        Map<String, List<String>> adj = new HashMap<>();
        adj.put("A", Arrays.asList("B", "C"));
        adj.put("B", Arrays.asList("D", "E"));
        adj.put("C", Arrays.asList("F"));
        adj.put("D", new ArrayList<>()); adj.put("E", new ArrayList<>()); adj.put("F", new ArrayList<>());

        Queue<String> q = new LinkedList<>(); Set<String> visited = new HashSet<>();
        q.add("A"); visited.add("A");
        while(!q.isEmpty()) {
            String u = q.poll(); System.out.print(u + " ");

        }
        System.out.println();
    }
}

How It Works

Performs Breadth-First Search on a graph starting from vertex A. BFS explores all neighbors at the current depth before moving deeper.

  • Uses a Queue (FIFO) to track the next vertex to visit
  • Uses a HashSet to prevent revisiting nodes
  • Traversal order: A → B → C → D → E → F (level by level)
Time Complexity: O(V + E)

Q7 Minimum Spanning Tree (Prim's & Kruskal's)

import java.util.*;

class Edge implements Comparable<Edge> {
    int src, dest, weight;
    public Edge(int src, int dest, int weight) {
        this.src = src; this.dest = dest; this.weight = weight;
    }
    public int compareTo(Edge compareEdge) {
        return this.weight - compareEdge.weight;
    }
}

public class MST {
    public static void main(String[] args) {
        int V = 6;
        List<Edge> edges = new ArrayList<>();
        edges.add(new Edge(0, 1, 4));
        edges.add(new Edge(0, 2, 4));
        edges.add(new Edge(1, 2, 2));
        edges.add(new Edge(1, 3, 5));
        edges.add(new Edge(2, 3, 5));
        edges.add(new Edge(2, 4, 9));
        edges.add(new Edge(3, 4, 7));
        edges.add(new Edge(3, 5, 6));
        edges.add(new Edge(4, 5, 3));
        
        System.out.println("--- Prim's Algorithm ---");
        primMST(V, edges);
        
        System.out.println("\n--- Kruskal's Algorithm ---");
        kruskalMST(V, edges);
    }
    
    static void primMST(int V, List<Edge> edges) {
        int[][] graph = new int[V][V];
        for (Edge e : edges) {
            graph[e.src][e.dest] = e.weight;
            graph[e.dest][e.src] = e.weight;
        }
        int[] parent = new int[V];
        int[] key = new int[V];
        boolean[] mstSet = new boolean[V];
        
        Arrays.fill(key, Integer.MAX_VALUE);
        key[0] = 0;
        parent[0] = -1;
        
        for (int count = 0; count < V - 1; count++) {
            int u = minKey(key, mstSet, V);
            mstSet[u] = true;
            for (int v = 0; v < V; v++) {
                if (graph[u][v] != 0 && !mstSet[v] && graph[u][v] < key[v]) {
                    parent[v] = u;
                    key[v] = graph[u][v];
                }
            }
        }
        
        int totalCost = 0;
        System.out.println("Edge \tWeight");
        for (int i = 1; i < V; i++) {
            System.out.println((char)(parent[i]+'A') + " - " + (char)(i+'A') + "\t" + graph[i][parent[i]]);
            totalCost += graph[i][parent[i]];
        }
        System.out.println("Total Cost: " + totalCost);
    }
    
    static int minKey(int[] key, boolean[] mstSet, int V) {
        int min = Integer.MAX_VALUE, min_index = -1;
        for (int v = 0; v < V; v++)
            if (!mstSet[v] && key[v] < min) {
                min = key[v];
                min_index = v;
            }
        return min_index;
    }
    
    static class Subset { int parent, rank; }
    static int find(Subset[] subsets, int i) {
        if (subsets[i].parent != i)
            subsets[i].parent = find(subsets, subsets[i].parent);
        return subsets[i].parent;
    }
    static void Union(Subset[] subsets, int x, int y) {
        int xroot = find(subsets, x);
        int yroot = find(subsets, y);
        if (subsets[xroot].rank < subsets[yroot].rank)
            subsets[xroot].parent = yroot;
        else if (subsets[xroot].rank > subsets[yroot].rank)
            subsets[yroot].parent = xroot;
        else {
            subsets[yroot].parent = xroot;
            subsets[xroot].rank++;
        }
    }
    
    static void kruskalMST(int V, List<Edge> edges) {
        Edge[] result = new Edge[V];
        int e = 0;
        int i = 0;
        Collections.sort(edges);
        
        Subset[] subsets = new Subset[V];
        for (i = 0; i < V; ++i) {
            subsets[i] = new Subset();
            subsets[i].parent = i;
            subsets[i].rank = 0;
        }
        
        i = 0;
        while (e < V - 1 && i < edges.size()) {
            Edge next_edge = edges.get(i++);
            int x = find(subsets, next_edge.src);
            int y = find(subsets, next_edge.dest);
            if (x != y) {
                result[e++] = next_edge;
                Union(subsets, x, y);
            }
        }
        
        int totalCost = 0;
        System.out.println("Edge \tWeight");
        for (i = 0; i < e; ++i) {
            System.out.println((char)(result[i].src+'A') + " - " + (char)(result[i].dest+'A') + "\t" + result[i].weight);
            totalCost += result[i].weight;
        }
        System.out.println("Total Cost: " + totalCost);
    }
}

📋 Program Output

--- Prim's Algorithm --- Edge Weight A - B 4 B - C 2 C - D 5 D - F 6 E - F 3 Total Cost: 20 --- Kruskal's Algorithm --- Edge Weight B - C 2 E - F 3 A - B 4 C - D 5 D - F 6 Total Cost: 20

How It Works

Finds the Minimum Spanning Tree of a weighted graph connecting 6 cities (A-F) using two algorithms.

  • Prim's: Starts from vertex A, greedily adds the cheapest edge connecting to an unvisited vertex
  • Kruskal's: Sorts all edges by weight, adds them if they don't form a cycle (uses Union-Find)
  • Both algorithms produce MST with total cost = 20
Prim's: O(V²) | Kruskal's: O(E log E)

public class RabinKarp {
    public static final int d = 256;
    public static void search(String pat, String txt, int q) {
        int M = pat.length(), N = txt.length(), p = 0, t = 0, h = 1;
        int spuriousHits = 0;
        for (int i = 0; i < M - 1; i++) h = (h * d) % q;
        for (int i = 0; i < M; i++) { p = (d * p + pat.charAt(i)) % q; t = (d * t + txt.charAt(i)) % q; }
        System.out.println("Hash of pattern: " + p);
        for (int i = 0; i <= N - M; i++) {
            if (p == t) {
                int j;
                for (j = 0; j < M; j++) if (txt.charAt(i+j) != pat.charAt(j)) break;
                if (j == M) System.out.println("Pattern found at index " + i);
                else { spuriousHits++; System.out.println("Spurious hit at index " + i); }
            }
            if (i < N - M) { t = (d*(t - txt.charAt(i)*h) + txt.charAt(i+M)) % q; if (t < 0) t += q; }
        }
        System.out.println("Spurious Hits: " + spuriousHits);
    }
    public static void main(String[] args) {

public class LCS {
    public static void main(String[] args) {
        String s1 = "AGGTAB", s2 = "GXTXAYB";
        char[] X = s1.toCharArray(), Y = s2.toCharArray();
        int m = X.length, n = Y.length;
        int[][] L = new int[m+1][n+1];
        for (int i = 0; i <= m; i++)
            for (int j = 0; j <= n; j++)
                if (i == 0 || j == 0) L[i][j] = 0;
                else if (X[i-1] == Y[j-1]) L[i][j] = L[i-1][j-1] + 1;
                else L[i][j] = Math.max(L[i-1][j], L[i][j-1]);
        System.out.println("LCS DP Table:");
        for(int i = 0; i <= m; i++) {
            for(int j = 0; j <= n; j++) System.out.print(L[i][j] + " ");
            System.out.println();
        }
        int index = L[m][n], temp = index;
        char[] lcs = new char[index];
        int i = m, j = n;
        while (i > 0 && j > 0) {
            if (X[i-1] == Y[j-1]) { lcs[--index] = X[i-1]; i--; j--; }
            else if (L[i-1][j] > L[i][j-1]) i--;
            else j--;
        }
        System.out.print("LCS: "); for(char c : lcs) System.out.print(c);
        System.out.println("\nLength: " + temp);
    }
}

How It Works

Finds the Longest Common Subsequence of "AGGTAB" and "GXTXAYB" using dynamic programming.

  • Builds a 2D DP table where L[i][j] = LCS length of first i chars of S1 and first j chars of S2
  • If characters match: L[i][j] = L[i-1][j-1] + 1
  • Otherwise: L[i][j] = max(L[i-1][j], L[i][j-1])
  • Backtracking through the table recovers the actual LCS: GTAB
Time Complexity: O(m × n)

public class StringMatchingCompare {
    public static void main(String[] args) {
        String text = "THIS IS A TEST TEXT FOR SPAM FILTERING WIN FREE OFFER WIN";
        String pattern = "WIN";
        System.out.println("Text: " + text); System.out.println("Pattern: " + pattern);
        long start = System.nanoTime(); naiveSearch(pattern, text);
        long naiveTime = System.nanoTime() - start;
        start = System.nanoTime(); rabinKarp(pattern, text, 101);
        long rkTime = System.nanoTime() - start;
        System.out.println("\nNaive Time (ns): " + naiveTime);
        System.out.println("Rabin-Karp Time (ns): " + rkTime);
    }
    static void naiveSearch(String pat, String txt) {
        int M = pat.length(), N = txt.length();
        for (int i = 0; i <= N - M; i++) {
            int j; for (j = 0; j < M; j++) if (txt.charAt(i+j) != pat.charAt(j)) break;
        }
    }
    static void rabinKarp(String pat, String txt, int q) {
        int d = 256, M = pat.length(), N = txt.length(), p = 0, t = 0, h = 1;
        for (int i = 0; i < M - 1; i++) h = (h * d) % q;
        for (int i = 0; i < M; i++) { p = (d*p + pat.charAt(i))%q; t = (d*t + txt.charAt(i))%q; }
        for (int i = 0; i <= N-M; i++) {
            if (p == t) { int j; for (j = 0; j < M; j++) if (txt.charAt(i+j) != pat.charAt(j)) break; }
            if (i < N-M) { t = (d*(t - txt.charAt(i)*h) + txt.charAt(i+M))%q; if (t<0) t+=q; }
        }
    }
}

How It Works

Compares Naïve String Matching vs Rabin-Karp for spam email detection.

  • Naïve: Checks every position character by character — O(N×M) worst case
  • Rabin-Karp: Uses rolling hash for O(1) window shifts — O(N+M) average
  • For large email datasets with multiple patterns, Rabin-Karp is significantly faster
  • Rabin-Karp can search multiple patterns simultaneously using a hash set
Naïve: O(NM) | Rabin-Karp: O(N+M) avg
